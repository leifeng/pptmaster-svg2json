Export types:
- MergeDecision = 'merge' | 'no-merge' | 'cautious-merge'
- SharedStyle = { fill?: string; stroke?: string; 'stroke-width'?: string; opacity?: string; 'fill-rule'?: string }
- BoundingBox = { left: number; top: number; width: number; height: number }

Export function extractCommonStyle(elements: SvgElement[]): SharedStyle | null
- Iterate styleKeys = ['fill', 'stroke', 'stroke-width', 'opacity', 'fill-rule'] as const
- For each key, check if ALL elements share the same value
- Return Partial<SharedStyle> with only common values, or null if no common styles

Export function boxesOverlap(a: BoundingBox, b: BoundingBox): boolean
- Return !(a.left + a.width <= b.left || b.left + b.width <= a.left || a.top + a.height <= b.top || b.top + b.height <= a.top)

Export function shouldMerge(paths: SvgElement[], groupAttrs: Record<string, string>): MergeDecision
- Step 1: Check shared style (D-30) - call extractCommonStyle(paths)
  - If no common styles AND not all fill="none" -> return 'no-merge'
- Step 2: Check no independent semantics
  - If any path has id attribute -> return 'no-merge'
  - If any path has clip-path attribute -> return 'no-merge'
  - If any path has aria-* attributes -> return 'no-merge'
- Step 3: Check fill="none" OR bounding box overlap
  - Get fill values (effective, considering inherited from group)
  - If ALL paths have fill="none" -> return 'merge' (per D-31 fill="none" default no overlap)
  - If ANY path has fill="none", skip that path in overlap check but continue
  - Compute bounding boxes for paths with fill != "none"
  - If no bounding boxes overlap -> return 'merge'
  - If bounding boxes overlap -> return 'cautious-merge'
- Step 4: Check evenodd (PATH-MERGE-04)
  - If any path has fill-rule="evenodd" -> return 'cautious-merge'
  - Note: D-32 conflict resolved - following PATH-MERGE-04 (requirement) over D-32 (pre-discussion assumption). evenodd paths get 'cautious-merge' not 'no-merge' per PATH-MERGE-04 which requires marking as "谨慎合并" not skipping entirely.
- Default: return 'cautious-merge'

Export function mergePaths(paths: SvgElement[], commonStyle: SharedStyle, groupTransform?: string): string
- For each path's d attribute, call toAbsolutePath(d) from path-utils.ts
- Join absolute paths with single space separator (per D-33)
- Return concatenated d string

Export function createMergedShapeElement(paths: SvgElement[], commonStyle: SharedStyle, groupTransform: string | undefined, viewBox: [number, number], index: number): PPTShapeElement
- Call mergePaths() to get merged d
- Compute bounding box from merged d using pathBoundingBox() (extended in Task 2)
- Calculate left/top/width/height from bounding box, scaled appropriately
- Apply groupTransform if present (use parseTransform from coordinate-mapper.ts)
- Return PPTShapeElement with:
  - id: uuidv4()
  - type: 'shape'
  - left, top, width, height: from scaled bounding box
  - rotate: from groupTransform if rotate present
  - viewBox: [width, height] in path coordinates
  - path: merged d string
  - fixedRatio: false
  - fill: commonStyle.fill || '#000000'
  - opacity: commonStyle.opacity ? parseFloat(commonStyle.opacity) : undefined
  - outline: if commonStyle.stroke present, parse stroke attributes
