    **Step 2 — 新增 findContentBounds() helper (src/index.ts line 180 之前)**

    在 `src/index.ts` 文件顶部(`findBackgroundGroup` 之前,line 89 之前)新增:

    ```ts
    /**
     * 扫描 SVG AST,返回第一个 `<g id="content-object-slot-template">` 的 data-pptx-bounds
     * 解析结果(空格分隔 4 数: x y w h)。深度优先遍历,遇到第一个匹配即返回。
     * 四数不足或非数字 → 返回 undefined。
     * D-TPL-04: 在 convertOne 内调用,不依赖 transformToPPTElement 签名。
     */
    function findContentBounds(root: SvgElement): { x: number; y: number; width: number; height: number } | undefined {
      // 深度优先遍历 root.children
      const stack: SvgElement[] = [root]
      while (stack.length > 0) {
        const node = stack.shift()!
        if (node.attrs.id === 'content-object-slot-template') {
          const bounds = node.attrs['data-pptx-bounds']
          if (typeof bounds !== 'string') return undefined
          // 严格空格分隔(/[x]+/),过滤空字符串
          const parts = bounds.trim().split(/[x]+/).filter(p => p.length > 0)
          if (parts.length !== 4) return undefined
          const [x, y, width, height] = parts.map(Number)
          if (parts.some(n => Number.isNaN(n))) return undefined
          return { x, y, width, height }
        }
        // 推入 children 到栈顶继续 DFS
        for (const child of node.children) stack.push(child)
      }
      return undefined
    }
    ```

    注意事项:
    - 严格空格分隔(`/x/`)匹配 D-TPL-02 描述的"纸面 PPTist 协议"
    - 不用 `parseFloat` 防止 `"98abc"` 这种前缀匹配被接受
    - 第一匹配的 slot 优先(TPL-CB-02 锁定"仅第一个");深度优先即满足文档顺序
    - 找不到 slot / 四数不足 / 含 NaN → 返回 `undefined`,调用方据此决定是否写字段
