    import { transformElement } from '../src/transformer/transform-to-pptel'

    function makeText(attrs: Record<string, string>): ReturnType&lt;typeof transformElement&gt; {
      const { SvgElement } = require('../src/types/ast')
      const el: SvgElement = { type: 'text', attrs, children: [], sourceIndex: 0 }
      return transformElement(el)
    }

    describe('text-anchor → text-align (TEXT-01)', () => {
      it('maps text-anchor="start" to text-align: left', () => {
        const result = makeText({ 'text-anchor': 'start', x: '0', y: '0' })
        expect(result.content).toContain('text-align: left')
      })

      it('maps text-anchor="middle" to text-align: center', () => {
        const result = makeText({ 'text-anchor': 'middle', x: '0', y: '0' })
        expect(result.content).toContain('text-align: center')
      })

      it('maps text-anchor="end" to text-align: right', () => {
        const result = makeText({ 'text-anchor': 'end', x: '0', y: '0' })
        expect(result.content).toContain('text-align: right')
      })

      it('omits text-align when text-anchor is absent', () => {
        const result = makeText({ x: '0', y: '0' })
        expect(result.content).not.toContain('text-align')
      })
    })
